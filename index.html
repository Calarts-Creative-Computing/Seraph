Seraph

Seraph – Open-Source Teensy MIDI Controller Platform. An open platform for developing Teensy-based MIDI controllers and interactive musical interfaces.

View on GitHub · Download ZIP

What's Inside

  • Seraph PCB design – A hardware platform for connecting various sensors and controls.
  • Demo code – Examples showing how to interface different sensors with the Seraph PCB.
  • Customization support – A flexible base for custom MIDI and interactive art projects.

Quick Start

  1. Solder your Teensy to headers so it can be removed and reused later.
  2. Install required Arduino libraries (most demos use the Teensy USBMIDI library).
  3. Set Teensy board mode to Serial + MIDI in the Arduino IDE.
  4. Upload a demo and start experimenting with interactive MIDI control.

Quickstart Tutorials & Wiring Diagrams

Connecting an Analog Sensor (No Pulldown Resistor)

For simple analog sensors like potentiometers that do not require pulldown resistors.

  1. Connect your power lead to any + (power) via.
  2. Connect your ground lead to any – (ground) via.
  3. Connect your data lead to any available analog input via, and update your code to match the pin you chose.

Connecting an Analog Sensor with a Pulldown Resistor

For sensors such as LDRs or FSRs that require a pulldown resistor.

  1. Connect your power lead to any + (power) via.
  2. Connect your data lead to your desired analog input via, and reflect this in your code.
  3. Place your pulldown resistor in the allocated resistor area on the PCB in the same row as your chosen analog input.

Connecting a Digital Sensor

For basic digital inputs like buttons that don't require pull resistors.

  1. Connect one lead of the button to any – (ground) via.
  2. Connect the other lead to the desired digital input via, and reflect this in your code.
  3. To add an LED, place it in an allocated LED slot and add a resistor in the same row.

Connecting an I²C Sensor

For I²C sensors such as IMUs or environmental sensors.

  1. Connect the SDA (Data) pin on your sensor to an SDA pin on Seraph.
  2. Connect the SCL (Clock) pin on your sensor to an SCL pin on Seraph.
  3. Connect the Power (VCC) pin to a + (power) via.
  4. Connect the Ground (GND) pin to a – (ground) via.

Seraph_PotentiometerDemo.ino 
//SERAPH POTENTIOMETER DEMO

#include <usb_midi.h>
#include <Wire.h>

int pot_pin = A9;
int pot_val_prev = 0;
int pot_val_lp = 0;

void setup() {
  Serial.begin(9600);
  usbMIDI.begin();
}

void loop() {
  int pot_val = map(analogRead(pot_pin), 1023, 0, 0, 127);
  pot_val_lp = (pot_val_prev + pot_val) / 2;

  Serial.print(" pot value:");
  Serial.print(pot_val);
  Serial.println();

  if (abs(pot_val_prev - pot_val) > 1) {
    usbMIDI.sendControlChange(19, pot_val_lp, 1);
  }

  while (usbMIDI.read()) {}
  pot_val_prev = pot_val;
  delay(1);
}

Seraph_ButtonDemo.ino 
//SERAPH BUTTON DEMO

#include <usb_midi.h>

int button_pin = 0;
int led_pin = 3;
int button_val_prev = 0;

void setup() {
  Serial.begin(9600);
  usbMIDI.begin();
}

void loop() {
  int button_val = map(digitalRead(button_pin), 0, 1, 0, 127);
  int button_state = digitalRead(button_pin);

  Serial.print(" button value:");
  Serial.print(button_val);
  Serial.println();

  if (button_val_prev != button_val) {
    usbMIDI.sendControlChange(20, button_val, 1);
    button_val_prev = button_val;
  }

  if (button_state == HIGH) {
    usbMIDI.sendNoteOn(60, 127, 1);
    digitalWrite(led_pin, HIGH);
  } else {
    usbMIDI.sendNoteOff(60, 127, 1);
    digitalWrite(led_pin, LOW);
  }

  while (usbMIDI.read()) {}
  button_val_prev = button_val;
  delay(1);
}
Seraph_6ButtonDemo.ino 
//SERAPH MULTIPLE BUTTON DEMO

#include <usb_midi.h>

int button1_pin = 0, button2_pin = 1, button3_pin = 2;
int button4_pin = 3, button5_pin = 4, button6_pin = 5;
int button1_val_prev = 0, button2_val_prev = 0, button3_val_prev = 0;
int button4_val_prev = 0, button5_val_prev = 0, button6_val_prev = 0;

void setup() {
  Serial.begin(9600);
  usbMIDI.begin();
}

void loop() {
  int button1_val = map(digitalRead(button1_pin), 0, 1, 0, 127);
  int button2_val = map(digitalRead(button2_pin), 0, 1, 0, 127);
  int button3_val = map(digitalRead(button3_pin), 0, 1, 0, 127);
  int button4_val = map(digitalRead(button4_pin), 0, 1, 0, 127);
  int button5_val = map(digitalRead(button5_pin), 0, 1, 0, 127);
  int button6_val = map(digitalRead(button6_pin), 0, 1, 0, 127);

  if (button1_val_prev != button1_val) { usbMIDI.sendControlChange(20, button1_val, 1); usbMIDI.sendNoteOn(60, 127, 1); } else { usbMIDI.sendNoteOff(60, 127, 1); }
  if (button2_val_prev != button2_val) { usbMIDI.sendControlChange(21, button2_val, 1); usbMIDI.sendNoteOn(64, 127, 1); } else { usbMIDI.sendNoteOff(64, 127, 1); }
  if (button3_val_prev != button3_val) { usbMIDI.sendControlChange(22, button3_val, 1); usbMIDI.sendNoteOn(67, 127, 1); } else { usbMIDI.sendNoteOff(67, 127, 1); }
  if (button4_val_prev != button4_val) { usbMIDI.sendControlChange(23, button4_val, 1); usbMIDI.sendNoteOn(71, 127, 1); } else { usbMIDI.sendNoteOff(71, 127, 1); }
  if (button5_val_prev != button5_val) { usbMIDI.sendControlChange(24, button5_val, 1); usbMIDI.sendNoteOn(72, 127, 1); } else { usbMIDI.sendNoteOff(72, 127, 1); }
  if (button6_val_prev != button6_val) { usbMIDI.sendControlChange(25, button6_val, 1); usbMIDI.sendNoteOn(76, 127, 1); } else { usbMIDI.sendNoteOff(76, 127, 1); }

  while (usbMIDI.read()) {}
  button1_val_prev = button1_val; button2_val_prev = button2_val; button3_val_prev = button3_val;
  button4_val_prev = button4_val; button5_val_prev = button5_val; button6_val_prev = button6_val;
  delay(1);
}
Seraph_AccelDemo.ino 
//SERAPH ACCELEROMETER DEMO

#include <usb_midi.h>
#include <Wire.h>

// Built-in Seraph accelerometer pins
int accelX_pin = A15, accelY_pin = A16, accelZ_pin = A17;
int accelX_val_prev = 0, accelY_val_prev = 0, accelZ_val_prev = 0;
int accelX_val_lp = 0, accelY_val_lp = 0, accelZ_val_lp = 0;

void setup() {
  Serial.begin(9600);
  usbMIDI.begin();
}

void loop() {
  int accelX_val = map(analogRead(accelX_pin), 1023, 0, 0, 127);
  int accelY_val = map(analogRead(accelY_pin), 1023, 0, 0, 127);
  int accelZ_val = map(analogRead(accelZ_pin), 1023, 0, 0, 127);

  accelX_val_lp = (accelX_val_prev + accelX_val) / 2;
  accelY_val_lp = (accelY_val_prev + accelY_val) / 2;
  accelZ_val_lp = (accelZ_val_prev + accelZ_val) / 2;

  if (accelX_val_lp > 10 || abs(accelX_val - accelX_val_prev) > 1) { usbMIDI.sendControlChange(16, accelX_val, 1); accelX_val_prev = accelX_val; }
  if (accelY_val_lp > 10 || abs(accelY_val - accelY_val_prev) > 1) { usbMIDI.sendControlChange(17, accelY_val, 1); accelY_val_prev = accelY_val; }
  if (accelZ_val_lp > 10 || abs(accelZ_val - accelZ_val_prev) > 1) { usbMIDI.sendControlChange(18, accelZ_val, 1); accelZ_val_prev = accelZ_val; }

  while (usbMIDI.read()) {}
  accelX_val_prev = accelX_val; accelY_val_prev = accelY_val; accelZ_val_prev = accelZ_val;
  delay(1);
}

//SERAPH ACCELEROMETER, BUTTON, and POTENTIOMETER DEMO

#include <usb_midi.h>
#include <Wire.h>

int accelX_pin = A15, accelY_pin = A16, accelZ_pin = A17;
int pot_pin = A9;
int button_pin = 0;
int accelX_val_prev = 0, accelY_val_prev = 0, accelZ_val_prev = 0;
int accelX_val_lp = 0, accelY_val_lp = 0, accelZ_val_lp = 0;
int pot_val_prev = 0, button_val_prev = 0;

void setup() {
  Serial.begin(9600);
  usbMIDI.begin();
}

void loop() {
  int accelX_val = map(analogRead(accelX_pin), 1023, 0, 0, 127);
  int accelY_val = map(analogRead(accelY_pin), 1023, 0, 0, 127);
  int accelZ_val = map(analogRead(accelZ_pin), 1023, 0, 0, 127);
  int pot_val    = map(analogRead(pot_pin),    1023, 0, 0, 127);
  int button_val = map(digitalRead(button_pin), 0, 1, 0, 127);

  accelX_val_lp = (accelX_val_prev + accelX_val) / 2;
  accelY_val_lp = (accelY_val_prev + accelY_val) / 2;
  accelZ_val_lp = (accelZ_val_prev + accelZ_val) / 2;

  if (accelX_val_lp > 10 || abs(accelX_val - accelX_val_prev) > 1) { usbMIDI.sendControlChange(16, accelX_val, 1); accelX_val_prev = accelX_val; }
  if (accelY_val_lp > 10 || abs(accelY_val - accelY_val_prev) > 1) { usbMIDI.sendControlChange(17, accelY_val, 1); accelY_val_prev = accelY_val; }
  if (accelZ_val_lp > 10 || abs(accelZ_val - accelZ_val_prev) > 1) { usbMIDI.sendControlChange(18, accelZ_val, 1); accelZ_val_prev = accelZ_val; }
  if (abs(pot_val_prev - pot_val) > 1)    { usbMIDI.sendControlChange(19, pot_val, 1); }
  if (button_val_prev != button_val)       { usbMIDI.sendControlChange(20, button_val, 1); button_val_prev = button_val; }

  while (usbMIDI.read()) {}
  accelX_val_prev = accelX_val; accelY_val_prev = accelY_val; accelZ_val_prev = accelZ_val;
  pot_val_prev = pot_val; button_val_prev = button_val;
  delay(1);
}
